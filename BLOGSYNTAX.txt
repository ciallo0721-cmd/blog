# BlockScript 语法速查卡 v1.0
# 用法：把下面内容保存为 {id}.blog，和 index.html（解码器）放同一文件夹

# ========== 文章头部（必写） ==========
[Title] 文章标题
[Date] 2026-04-15
[Author] 作者名
[Tag] 标签1, 标签2, 标签3

[正文开始]
# ↑ 这行必须有，上面是头部，下面是正文


# ========== 正文写法 ==========

# --- 标题 ---
[H1] 一级标题（最大）
[H2] 二级标题
[H3] 三级标题

# --- 段落 ---
直接写文字就是一段
空一行再写就是新段落

# --- 文字样式 ---
**加粗文字**
*斜体文字*
`代码或文件名`

# --- 链接 ---
[链接文字](https://网址)

# --- 列表 ---
[ul]
- 项目1
- 项目2
- 项目3
[/ul]

[ol]
1. 第一步
2. 第二步
3. 第三步
[/ol]

# --- 引用 ---
[quote]
引用内容写这里
可以多行
[/quote]

# --- 代码块 ---
[Code lang=python]
def hello():
    print("Hello")
[/Code]
# lang= 后面可以写：python, javascript, html, css, bash 等

# --- 提示框 ---
[Alert type=info]
普通提示信息
[/Alert]

[Alert type=success]
成功提示
[/Alert]

[Alert type=warning]
警告提示
[/Alert]

[Alert type=danger]
危险提示
[/Alert]

# --- 按钮 ---
[Btn href=https://网址 color=primary]按钮文字[/Btn]
# color=primary  蓝色按钮
# color=secondary  灰色按钮

# --- 表格 ---
[Table]
列标题1|列标题2|列标题3
内容1|内容2|内容3
内容4|内容5|内容6
[/Table]

# --- 分割线 ---
[HR]

# --- 直接写 HTML（高级） ---
[HTML]
<img src="./screenshot.png" style="max-width:100%">
[/HTML]
